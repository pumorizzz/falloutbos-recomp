cmake_minimum_required(VERSION 3.20)
project(fbos_game C)

# Fallout: Brotherhood of Steel (Xbox) - Recompiled game executable

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# ── Keep gen/ in step with the toolkit's codegen ──────────────
# gen/ is gitignored, so a lifter fix pulled with the toolkit never reaches
# the game unless the C is generated again. Any codegen source newer than
# gen/ re-runs ./regen.sh here; listing them as configure dependencies makes
# a plain `cmake --build` re-configure (and so get here) when one changes.
option(FBOS_AUTO_REGEN "Re-run regen.sh when the toolkit's codegen changes" ON)
if(FBOS_AUTO_REGEN)
    get_filename_component(_fbos_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    set(_gen_stamp "${CMAKE_CURRENT_SOURCE_DIR}/recomp/gen/.regen_stamp")
    file(GLOB_RECURSE _codegen_inputs
        "${XBOXRECOMP_DIR}/tools/recomp/*.py"   "${XBOXRECOMP_DIR}/tools/recomp/*.json"
        "${XBOXRECOMP_DIR}/tools/func_id/*.py"  "${XBOXRECOMP_DIR}/tools/func_id/*.json"
        "${XBOXRECOMP_DIR}/tools/abi_analysis/*.py")
    list(FILTER _codegen_inputs EXCLUDE REGEX "/test_[^/]*$|/__pycache__/")
    # the port's own codegen: aspect.py lifts widescreen.py's sites as table reads
    list(APPEND _codegen_inputs "${_fbos_root}/tools/aspect.py" "${_fbos_root}/tools/widescreen.py")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_codegen_inputs})
    set(_stale "")
    set(_no_gen OFF)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/recomp/gen/recomp_funcs.h"
       AND NOT EXISTS "${_gen_stamp}")
        # A fresh clone: no generated C at all, and without it the link fails.
        set(_stale "there is no generated C yet")
        set(_no_gen ON)
    elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/recomp/gen/recomp_funcs.h"
       AND NOT EXISTS "${_gen_stamp}")
        set(_stale "it has no regen stamp yet")
    elseif(EXISTS "${_gen_stamp}")
        foreach(_f IN LISTS _codegen_inputs)
            if("${_f}" IS_NEWER_THAN "${_gen_stamp}")
                set(_stale "${_f} is newer")
                break()
            endif()
        endforeach()
    endif()
    if(_no_gen AND NOT EXISTS "${_fbos_root}/game/default.xbe")
        message(FATAL_ERROR "src/game/recomp/gen/ is empty and there is no game/default.xbe "
            "to generate it from. Unpack your own disc into game/ (Setup.cmd does it), "
            "then configure again.")
    elseif(_stale AND NOT EXISTS "${_fbos_root}/game/default.xbe")
        message(WARNING "gen/ is stale (${_stale}), but there is no game/default.xbe to regenerate from")
    elseif(_stale)
        # Git's bash, as tools/setup.ps1 uses: a bare `bash` on PATH can be
        # WSL's, which cannot run py -3 on these paths.
        find_package(Git REQUIRED)
        get_filename_component(_git_root "${GIT_EXECUTABLE}" DIRECTORY)
        get_filename_component(_git_root "${_git_root}" DIRECTORY)
        find_program(FBOS_BASH bash HINTS "${_git_root}/bin" "${_git_root}/../bin"
                     NO_DEFAULT_PATH REQUIRED)
        message(STATUS "gen/ is stale (${_stale}): running regen.sh (a few minutes)")
        execute_process(
            COMMAND "${CMAKE_COMMAND}" -E env "RECOMP=${XBOXRECOMP_DIR}" "${FBOS_BASH}" ./regen.sh
            WORKING_DIRECTORY "${_fbos_root}"
            RESULT_VARIABLE _regen_status)
        if(NOT _regen_status EQUAL 0)
            message(FATAL_ERROR "regen.sh failed (${_regen_status}); gen/ is stale")
        endif()
    endif()
endif()

# ── Recompiled function sources (auto-generated) ──────────────
# Generated by: py -3 -m tools.recomp <xbe> --split 1000
# CONFIGURE_DEPENDS: the chunk count changes whenever the function count
# does, and a plain GLOB is evaluated once at configure time. Recovering
# 2,425 more functions took this from 46 chunks to 48, and the two new
# files were simply not compiled -- which surfaces as LNK2019 on symbols
# that are visibly present in the generated sources.
file(GLOB RECOMP_GEN_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/recomp/gen/recomp_*.c"
)

# Game executable sources
set(GAME_SOURCES
    main.c
    # Per-game override hook, from templates/new-game. Unlike recomp_types.h
    # this one is *meant* to be edited per title: it is where a function the
    # lifter got wrong gets replaced by hand.
    recomp/recomp_manual.c
    ${RECOMP_GEN_SOURCES}
)

# Create console application (for debugging; switch to WIN32 for release)
add_executable(fbos ${GAME_SOURCES})

# Link compatibility layers
target_link_libraries(fbos PRIVATE
    xbox_kernel
    xbox_d3d8
    xbox_dsound
    xbox_input
    xbox_usb
    xbox_video
    xbox_apu
)

# System libraries
target_link_libraries(fbos PRIVATE
    kernel32
    user32
    gdi32
    # timeBeginPeriod, for the profiler's 2 ms sample interval.
    winmm
    # Resolves the host address in a fault report back to the generated
    # function's symbol, i.e. to the guest function that faulted.
    dbghelp
)

# Include paths
target_include_directories(fbos PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/recomp
    ${CMAKE_CURRENT_SOURCE_DIR}/recomp/gen
    ${CMAKE_SOURCE_DIR}/src
    # recomp_types.h et al come from the toolkit template directory rather than
    # a copy in this repo. Bloodwake and Burnout 3 each carry a copy that has
    # since diverged from the template by hundreds of lines; Halo and Wreckless
    # carry copies that are still byte-identical, which is the same latent
    # problem. Including the source of truth means this repo cannot drift.
    ${XBOXRECOMP_DIR}/templates/runtime
)

# Compiler options
if(MSVC)
    target_compile_options(fbos PRIVATE
        /W4
        /WX-
        /O2
        # cmake --build -j maps to MSBuild /m, which parallelises across
        # *projects*. Every generated chunk lives in this one project, so
        # without /MP the compiler never forks and a full rebuild of 480k
        # lines runs single-threaded.
        #
        # Bare /MP means one cl.exe per logical core. Each one holds a whole
        # generated chunk -- some are 2.8 MB of C -- in memory at /O2, so on
        # a 12-core box that is 12 multi-gigabyte compilers at once and the
        # machine pages itself to death. Cap it, and leave cores for whoever
        # is using the desktop. Raise with -DFBOS_COMPILE_JOBS=N.
        /MP${FBOS_COMPILE_JOBS}
    )
    # Reduce warning noise on generated/recompiled code
    set_source_files_properties(${RECOMP_GEN_SOURCES} PROPERTIES
        COMPILE_FLAGS "/w"
    )
    target_compile_definitions(fbos PRIVATE
        _CRT_SECURE_NO_WARNINGS
    )
    # Disable ASLR so we can map Xbox memory at fixed addresses
    # Large stack: recompiled code uses deep native call chains
    target_link_options(fbos PRIVATE
        /DYNAMICBASE:NO
        /FIXED:NO
        /MAP
        /DEBUG
        /STACK:16777216
        # Do not fold identical functions together.
        #
        # Release defaults to /OPT:ICF, and this target is 3,500 generated
        # functions of which a great many are byte-identical -- one-line
        # forwarders, vcall thunks, trivial accessors. ICF collapses them and
        # the PDB then maps that one address to whichever name won, so a fault
        # report names a function that had nothing to do with the crash. Every
        # instrument built on the symbol -- the fault handler, the guest
        # backtrace, the watchdog -- inherits the lie.
        #
        # Costs image size and nothing else; /OPT:REF still drops what is
        # genuinely unreachable.
        /OPT:NOICF
        /OPT:REF
    )
elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU" OR CMAKE_C_COMPILER_ID STREQUAL "Clang")
    target_compile_options(fbos PRIVATE
        -Wall -Wextra -Wno-unused-parameter -O2
    )
    set_source_files_properties(${RECOMP_GEN_SOURCES} PROPERTIES
        COMPILE_FLAGS "-w"
    )
endif()

# Debug info
target_compile_definitions(fbos PRIVATE
    $<$<CONFIG:Debug>:_DEBUG>
)

# Output to project root for easy testing
set_target_properties(fbos PROPERTIES
    OUTPUT_NAME fobos                   # as the game names its saves (fobos.sav)
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin"
    RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/bin"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin"
)
